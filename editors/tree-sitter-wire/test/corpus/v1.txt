  <- label: Qubit
  -> label: Qubit = @quantum.rz { payload = label; cfg = { inherit angle; }; };

form turn(label: PortLabel) = {
  node rz = rz_gate(label, 1.5707963267948966);
  node sx = qubit_gate(label, quantum_sx);
  let path = rz => sx;

  path;
};

form eraser(phase: Value, marker: Graph) = {
  node phase_shift = rz_gate(q, phase);
  let inner = turn(q);
  let tagged = marker;

  (inner => phase_shift) <> tagged;
};
